# Dependencies for the REST API service layer (project/api).
#
# The RAG core keeps its own pinning in requirements.txt; this file only adds
# what the API needs on top of it. Install BOTH:
#
#   pip install -r requirements.txt          # RAG core (torch, langgraph, gradio, ...)
#   pip install -r requirements-api.txt      # API service layer
#
# fastapi/uvicorn/SQLAlchemy are already pulled in transitively by
# langchain-deepseek/ragas, but they are pinned here so the API layer can be
# installed on its own and cannot drift underneath us.

# Multipart form parsing for POST /api/v1/documents (file upload).

# PostgreSQL persistence: conversations / messages / documents.

alembic==1.20.0

langgraph-checkpoint-postgres==3.0.5

# Exponential backoff around LLM calls.
